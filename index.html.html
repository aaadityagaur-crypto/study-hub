Popular C Code Snippets

Check Prime Number

#include <stdio.h>
int main() {
    int n = 11, isPrime = 1;
    for(int i = 2; i <= n/2; i++) {
        if(n % i == 0) { isPrime = 0; break; }
    }
    if(isPrime) printf("Prime");
    else printf("Not Prime");
    return 0;
}

Factorial using Recursion

#include <stdio.h>
long factorial(int n) {
    if (n == 0) return 1;
    else return n * factorial(n - 1);
}
int main() {
    printf("Factorial: %ld", factorial(5));
    return 0;
}
